Critical / CVSS 9.1 V4: パスワード平文保存

脆弱なコード

// 平文で比較
if (user.password !== password) {
  return res.status(401)...
}
// 平文のまま保存
.run(email, password, name)

修正後

// bcryptでハッシュ比較
const isValid = await bcrypt
  .compare(password, user.password)
// ハッシュ化して保存
const hash = await bcrypt
  .hash(password, 12)
.run(email, hash, name)

V3（SQLインジェクション）と連鎖すると、UNION SELECTで全パスワードを平文で窃取可能。

Medium / CVSS 5.3 V7: エラーメッセージ情報漏洩

脆弱なコード

return res.status(500).json({
  error: "取得に失敗しました",
  detail: err.message,
  stack: err.stack, // DB構造が漏洩
})
// ログにパスワード出力
console.error({ email, password })

修正後

// 汎用メッセージのみ返す
return res.status(500).json({
  error: "取得に失敗しました"
})
// パスワードはログに含めない
console.error({ email,
  error: err.message })

スタックトレースからDB構造、ファイルパス、ライブラリバージョンが攻撃者に渡る。

Medium / CVSS 6.5 V8: CSRF対策なし

脆弱なコード

// 空実装 -- 何もしていない

  _req, _res, next

// csrf-csrf で実装

// ログイン処理
export async function loginHandler(req: Request, res: Response) {
  const { email, password } = req.body

  if (!email || !password) {
    return res.status(400).json({ error: "メールアドレスとパスワードは必須です" })
  }

  try {
    const db = getDb()

    // [V3] SQLインジェクション -- CWE-89
    // 文字列結合でクエリを組み立てているため、任意のSQLを注入できる
    const query = `SELECT * FROM users WHERE email = '${email}'`
    const user = db.prepare(query).get() as
      | { id: number; email: string; password: string; name: string }
      | undefined

    if (!user) {
      return res.status(401).json({ error: "認証に失敗しました" })
    }

    // [V4] パスワードを平文で比較 -- CWE-256
    // bcrypt等でハッシュ化して保存・比較すべき
    if (user.password !== password) {
      return res.status(401).json({ error: "認証に失敗しました" })
    }

    // JWTトークン発行
    const token = jwt.sign(
      { userId: user.id, email: user.email },
      JWT_SECRET,
      { expiresIn: TOKEN_EXPIRY }
    )

    return res.json({
      token,
      user: { id: user.id, email: user.email, name: user.name },
    })
  } catch (error) {
    console.error("ログインエラー:", error)
    return res.status(500).json({ error: "内部エラーが発生しました" })
  }
}

// ユーザー登録処理
export async function registerHandler(req: Request, res: Response) {
  const { email, password, name } = req.body

  if (!email || !password || !name) {
    return res.status(400).json({ error: "全項目の入力が必要です" })
  }

  try {
    const db = getDb()

    // 既存ユーザーチェック（V3と同根のSQLインジェクション）
    const existing = db
      .prepare(`SELECT id FROM users WHERE email = '${email}'`)
      .get()

    if (existing) {
      return res.status(409).json({ error: "このメールアドレスは既に登録されています" })
    }

    // [V4と同根] パスワードを平文のまま保存
    const result = db
      .prepare("INSERT INTO users (email, password, name) VALUES (?, ?, ?)")
      .run(email, password, name)

    const token = jwt.sign(
      { userId: result.lastInsertRowid, email },
      JWT_SECRET,
      { expiresIn: TOKEN_EXPIRY }
    )

    return res.status(201).json({
      token,
      user: { id: result.lastInsertRowid, email, name },
    })
  } catch (error) {
    console.error("登録エラー:", error)
    return res.status(500).json({ error: "内部エラーが発生しました" })
  }
}

import { Request, Response, NextFunction } from "express"
import { verifyToken } from "./auth"

// 認証済みリクエストの型拡張
interface AuthenticatedRequest extends Request {
  user?: { userId: number; email: string }
}

// JWT認証ミドルウェア
export function authMiddleware(
  req: AuthenticatedRequest,
  res: Response,
  next: NextFunction
) {
  const authHeader = req.headers.authorization

  if (!authHeader || !authHeader.startsWith("Bearer ")) {
    return res.status(401).json({ error: "認証トークンが必要です" })
  }

  const token = authHeader.split(" ")[1]

  try {
    const decoded = verifyToken(token)
    req.user = decoded
    next()
  } catch (error) {
    return res.status(401).json({ error: "無効なトークンです" })
  }
}

export function initDb(): void {

  // WALモード有効化（パフォーマンス向上）

      email TEXT UNIQUE NOT NULL,
      password TEXT NOT NULL,
      name TEXT NOT NULL,
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP

  `)

  // デモ用初期データ挿入

    count: number
  }

  if (userCount.count === 0) {

    // デモ用ダミーデータ（本物のパスワードではない）

脆弱性の連鎖 -- 単体より組み合わせが危険

個別にはMediumやHighでも、連鎖で被害規模が跳ね上がるパターンが3つある。

連鎖パターン 1

UNION SELECTで取得したパスワードが平文。クラック不要で即時悪用可能。ハッシュ化されていればクラック時間が稼げるが、平文ではその猶予がない。

連鎖パターン 2

V10 + V4

ファイルパーミッション不備 + 平文パスワード

ネットワーク経由の攻撃すら不要。同一マシンの別プロセスがsqlite3でDBを開き、SELECT一発で全認証情報を取得できる。

連鎖パターン 3

V5 + V1

JWTシークレット既知 + CORS全開放

攻撃者がシークレットを知っていれば任意のトークンを発行でき、CORSが全開放なので任意のオリジンからAPIを呼べる。認証の仕組み全体が無力化する。
